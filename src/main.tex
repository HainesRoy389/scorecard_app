% Auto-colored delta:
% - Positive: strings beginning with '+' (except exact '+0' or '+0%')
% - Negative: strings beginning with '-' (except exact '-0' or '-0%')
% - Neutral: '0', '0%', '+0', '+0%', '-0', '-0%'
\newcommand{\autoD}[1]{%
	\IfStrEq{#1}{0}{\neuD{#1}}{%
	\IfStrEq{#1}{0\%}{\neuD{#1}}{%
	\IfStrEq{#1}{+0}{\neuD{#1}}{%
	\IfStrEq{#1}{+0\%}{\neuD{#1}}{%
	\IfStrEq{#1}{-0}{\neuD{#1}}{%
	\IfStrEq{#1}{-0\%}{\neuD{#1}}{%
	\IfBeginWith{#1}{+}{\posD{#1}}{%
	\IfBeginWith{#1}{-}{\negD{#1}}{\neuD{#1}}%
	}}}}}}}%
}
